Chapter 1 – String Programming

Q001String

Reverse a string using Java 8 Streams

Reverse characters without using StringBuilder in the Java 8 approach.

Complexity: O(n) time. Java 8 creates an output string of O(n) space; the Java 7 char-array implementation uses O(n) mutable storage.
Sample input / test data
String s = "hello";
Expected output
olleh
Java 8 complete runnable program
import java.util.*;
import java.util.stream.*;
import java.util.function.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class ReverseStringJava8 {

    public static void main(String[] args) {
        reverseString();
    }

private static void reverseString() {
    String s = "hello";
    String reversed = IntStream.range(0, s.length())
        .mapToObj(i -> String.valueOf(s.charAt(s.length() - 1 - i)))
        .collect(Collectors.joining());
    System.out.println(reversed);
}
}
Java 7 complete runnable program
import java.util.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class ReverseStringJava7 {

    public static void main(String[] args) {
        reverseString();
    }

private static void reverseString() {
    String s = "hello";
    char[] chars = s.toCharArray();
    for (int i = 0, j = chars.length - 1; i < j; i++, j--) {
        char temp = chars[i]; chars[i] = chars[j]; chars[j] = temp;
    }
    System.out.println(new String(chars));
}
}
Explanation and edge cases

Approach: The Java 8 solution creates indexes from zero to length minus one and reads the source characters in reverse order. The Java 7 solution swaps symmetric characters with two pointers. Both operate on UTF-16 char values, not full grapheme clusters.

Edge cases: Empty and one-character strings return unchanged. Null is outside this example contract and should be rejected before calling the method. Supplementary Unicode characters require code-point-aware reversal.

Interview follow-up: How would you reverse by Unicode code point? How does StringBuilder.reverse() differ from this implementation?

Complexity: O(n) time. Java 8 creates an output string of O(n) space; the Java 7 char-array implementation uses O(n) mutable storage.

Q002String

Reverse each word in a sentence

Reverse each token while retaining word order.

Complexity: O(n) time and O(n) output/intermediate space.
Sample input / test data
String sentence = "Java   is fun";
Expected output
avaJ si nuf
Java 8 complete runnable program
import java.util.*;
import java.util.stream.*;
import java.util.function.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class ReverseEachWordInSentenceJava8 {

    public static void main(String[] args) {
        reverseEachWordInSentence();
    }

private static void reverseEachWordInSentence() {
    String sentence = "Java   is fun";
    String result = sentence.trim().isEmpty() ? "" :
        Arrays.stream(sentence.trim().split("\\s+"))
            .map(word -> new StringBuilder(word).reverse().toString())
            .collect(Collectors.joining(" "));
    System.out.println(result);
}
}
Java 7 complete runnable program
import java.util.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class ReverseEachWordInSentenceJava7 {

    public static void main(String[] args) {
        reverseEachWordInSentence();
    }

private static void reverseEachWordInSentence() {
    String sentence = "Java   is fun";
    StringBuilder result = new StringBuilder();
    String trimmed = sentence.trim();
    if (!trimmed.isEmpty()) {
        String[] words = trimmed.split("\\s+");
        for (int i = 0; i < words.length; i++) {
            if (i > 0) result.append(" ");
            result.append(new StringBuilder(words[i]).reverse());
        }
    }
    System.out.println(result.toString());
}
}
Explanation and edge cases

Approach: Trim the sentence, split on one-or-more whitespace characters, reverse each token, and join tokens with one space. This deliberately normalizes repeated whitespace instead of preserving exact spacing.

Edge cases: Empty or whitespace-only input produces an empty string. Null must be rejected. If preserving original whitespace positions is required, tokenize whitespace and words separately.

Interview follow-up: How would you preserve multiple spaces and tabs exactly? What is the complexity if StringBuilder is used for each word?

Complexity: O(n) time and O(n) output/intermediate space.

Q003String

Find duplicate characters in a string

Return characters that occur more than once.

Complexity: O(n) expected time and O(k) space for k distinct char values.
Sample input / test data
String s = "apple";
Expected output
[p]
Java 8 complete runnable program
import java.util.*;
import java.util.stream.*;
import java.util.function.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class FindDuplicateCharactersInStringJava8 {

    public static void main(String[] args) {
        findDuplicateCharactersInString();
    }

private static void findDuplicateCharactersInString() {
    String s = "apple";
    Map<Character, Long> frequency = s.chars().mapToObj(ch -> (char) ch)
        .collect(Collectors.groupingBy(Function.identity(), LinkedHashMap::new, Collectors.counting()));
    List<Character> duplicates = frequency.entrySet().stream()
        .filter(e -> e.getValue() > 1).map(Map.Entry::getKey).collect(Collectors.toList());
    System.out.println(duplicates);
}
}
Java 7 complete runnable program
import java.util.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class FindDuplicateCharactersInStringJava7 {

    public static void main(String[] args) {
        findDuplicateCharactersInString();
    }

private static void findDuplicateCharactersInString() {
    String s = "apple";
    Map<Character, Integer> frequency = new LinkedHashMap<Character, Integer>();
    for (char ch : s.toCharArray()) frequency.put(ch, frequency.containsKey(ch) ? frequency.get(ch) + 1 : 1);
    List<Character> duplicates = new ArrayList<Character>();
    for (Map.Entry<Character, Integer> e : frequency.entrySet()) if (e.getValue() > 1) duplicates.add(e.getKey());
    System.out.println(duplicates);
}
}
Explanation and edge cases

Approach: Build a frequency map in first-appearance order, then collect keys whose count exceeds one. LinkedHashMap makes output ordering deterministic.

Edge cases: No duplicates returns []. Repeated characters appear only once in the result. This version counts UTF-16 char units; use code points for supplementary characters.

Interview follow-up: Can you find duplicates in one pass? What changes if output order is irrelevant?

Complexity: O(n) expected time and O(k) space for k distinct char values.

Q006String

Find the first repeated character

Return the first character whose second occurrence is encountered.

Complexity: O(n) expected time and O(k) space.
Sample input / test data
String s = "swiss";
Expected output
s
Java 8 complete runnable program
import java.util.*;
import java.util.stream.*;
import java.util.function.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class FindFirstRepeatedCharacterJava8 {

    public static void main(String[] args) {
        findFirstRepeatedCharacter();
    }

private static void findFirstRepeatedCharacter() {
    String s = "swiss";
    Set<Character> seen = new HashSet<Character>();
    Character repeated = s.chars().mapToObj(ch -> (char) ch)
        .filter(ch -> !seen.add(ch)).findFirst().orElse(null);
    System.out.println(repeated);
}
}
Java 7 complete runnable program
import java.util.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class FindFirstRepeatedCharacterJava7 {

    public static void main(String[] args) {
        findFirstRepeatedCharacter();
    }

private static void findFirstRepeatedCharacter() {
    String s = "swiss";
    Set<Character> seen = new HashSet<Character>();
    Character repeated = null;
    for (char ch : s.toCharArray()) if (!seen.add(ch)) { repeated = ch; break; }
    System.out.println(repeated);
}
}
Explanation and edge cases

Approach: Scan characters from left to right and maintain a set of previously seen values. The first failed add identifies the character whose second occurrence is encountered earliest. The Java 8 stream is sequential because the set is stateful.

Edge cases: No repeated character returns null. Do not convert this stream to parallel: the mutable seen set and encounter-order requirement make that unsafe.

Interview follow-up: How does this differ from the first character whose total frequency is greater than one? Can LinkedHashSet help?

Complexity: O(n) expected time and O(k) space.

Check whether a string is a palindrome

Compare characters from both ends.

Complexity: O(n) time and O(1) auxiliary space.
Sample input / test data
String s = "level";
Expected output
true
Java 8 complete runnable program
import java.util.*;
import java.util.stream.*;
import java.util.function.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class CheckWhetherStringIsPalindromeJava8 {

    public static void main(String[] args) {
        checkWhetherStringIsPalindrome();
    }

private static void checkWhetherStringIsPalindrome() {
    String s = "level";
    boolean palindrome = IntStream.range(0, s.length() / 2)
        .allMatch(i -> s.charAt(i) == s.charAt(s.length() - 1 - i));
    System.out.println(palindrome);
}
}
Java 7 complete runnable program
import java.util.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class CheckWhetherStringIsPalindromeJava7 {

    public static void main(String[] args) {
        checkWhetherStringIsPalindrome();
    }

private static void checkWhetherStringIsPalindrome() {
    String s = "level";
    boolean palindrome = true;
    for (int i = 0, j = s.length() - 1; i < j; i++, j--)
        if (s.charAt(i) != s.charAt(j)) { palindrome = false; break; }
    System.out.println(palindrome);
}
}
Explanation and edge cases

Approach: Only the first half needs checking. Each character at index i must equal the mirrored character at length minus one minus i. allMatch stops on the first mismatch.

Edge cases: Empty and one-character strings are palindromes. This is an exact, case-sensitive comparison; punctuation and case normalization are handled in S018.

Interview follow-up: How do you make this case-insensitive? How does the answer change for Unicode code points?

Complexity: O(n) time and O(1) auxiliary space.

Q010String

Find maximum occurring character

Return the character with highest frequency; ties use first appearance.

Complexity: O(n) expected time and O(k) space.
Sample input / test data
String s = "hello";
Expected output
l (frequency=2)

import java.util.stream.*;
import java.util.function.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class FindMaximumOccurringCharacterJava8 {

    public static void main(String[] args) {

    Character most = counts.entrySet().stream()
        .max((a, b) -> Long.compare(a.getValue(), b.getValue()))
        .map(Map.Entry::getKey).orElse(null);
    System.out.println(most == null ? "null" : most + " (frequency=" + counts.get(most) + ")");
}
}
Java 7 complete runnable program
import java.util.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class FindMaximumOccurringCharacterJava7 {

    public static void main(String[] args) {
        findMaximumOccurringCharacter();
    }

private static void findMaximumOccurringCharacter() {
    String s = "hello";
    Map<Character, Integer> counts = new LinkedHashMap<Character, Integer>();
    for (char ch : s.toCharArray()) counts.put(ch, counts.containsKey(ch) ? counts.get(ch) + 1 : 1);
    Character most = null; int max = 0;
    for (Map.Entry<Character, Integer> e : counts.entrySet()) if (e.getValue() > max) { most = e.getKey(); max = e.getValue(); }

Explanation and edge cases

Edge cases: Empty input returns null. If multiple characters share the maximum frequency, the earliest distinct character in the source wins.

Interview follow-up: How would you return all characters tied for maximum frequency? Can a single pass find the winner?

Complexity: O(n) expected time and O(k) space.

Q011String

Find second most frequently occurring character

Rank distinct characters by frequency; ties follow first appearance.

Complexity: O(n + k log k) time and O(k) space for k distinct characters.
Sample input / test data
String s = "aabbbc";
Expected output
a (frequency=2)
Java 8 complete runnable program
import java.util.*;
import java.util.stream.*;
import java.util.function.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class FindSecondMostFrequentlyOccurringCharacterJava8 {

    public static void main(String[] args) {
        findSecondMostFrequentlyOccurringCharacter();
    }

private static void findSecondMostFrequentlyOccurringCharacter() {
    String s = "aabbbc";
    Map<Character, Long> counts = s.chars().mapToObj(ch -> (char) ch)
        .collect(Collectors.groupingBy(Function.identity(), LinkedHashMap::new, Collectors.counting()));
    Character second = counts.entrySet().stream()
        .sorted((a, b) -> Long.compare(b.getValue(), a.getValue()))
        .skip(1).map(Map.Entry::getKey).findFirst().orElse(null);
    System.out.println(second == null ? "null" : second + " (frequency=" + counts.get(second) + ")");
}
}
Java 7 complete runnable program
import java.util.*;
import java.math.*;
import java.util.regex.*;
import java.text.*;

public class FindSecondMostFrequentlyOccurringCharacterJava7 {

    public static void main(String[] args) {
        findSecondMostFrequentlyOccurringCharacter();
    }

private static void findSecondMostFrequentlyOccurringCharacter() {
    String s = "aabbbc";
    Map<Character, Integer> counts = new LinkedHashMap<Character, Integer>();
    for (char ch : s.toCharArray()) counts.put(ch, counts.containsKey(ch) ? counts.get(ch) + 1 : 1);
    List<Character> keys = new ArrayList<Character>(counts.keySet());
    Collections.sort(keys, new Comparator<Character>() {
        public int compare(Character a, Character b) { return Integer.compare(counts.get(b), counts.get(a)); }
    });
    Character second = keys.size() > 1 ? keys.get(1) : null;
    System.out.println(second == null ? "null" : second + " (frequency=" + counts.get(second) + ")");
}
}
Explanation and edge cases

Approach: Count each distinct character, then sort by descending frequency. The ordered map and stable sort preserve first appearance when frequencies tie. The second entry in the ranking is returned; this is the second-ranked distinct character, not the second occurrence.

Edge cases: Fewer than two distinct characters returns null. For aabbbc the frequencies are b=3, a=2, c=1, so the correct answer is a (not b).

Interview follow-up: How would you return the top K characters? How can a heap reduce sorting cost?

Complexity: O(n + k log k) time and O(k) space for k distinct characters.

    for (char c : s.toCharArray()) if (Character.isLetterOrDigit(c)) cleaned.append(Character.toLowerCase(c));

    for (int i = 0; i < cleaned.length(); i++) result[i] = cleaned.charAt(i);
    return result;
}
